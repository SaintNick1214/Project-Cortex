/** Trusted finite-code route fences, not kernel isolation or protection from arbitrary code. */
import * as fs from 'node:fs';import * as path from 'node:path';
const reject=code=>{throw Error(code);};
export function routePolicy({roots,readFiles=[],readDirectories=[],effects=[],clock=Date.now}){
 const owned=roots.map(p=>path.resolve(p));const reads=new Set(readFiles.map(p=>path.resolve(p))),dirs=new Set(readDirectories.map(p=>path.resolve(p))),fds=new Map();
 const under=(p,r)=>p===r||p.startsWith(r+path.sep);
 function normalized(p){if(typeof p!=='string'||!path.isAbsolute(p)||p!==path.resolve(p)||p.includes('\0'))reject('QA_OUTPUT_PATH');return p;}
 function ancestors(p){let q=path.dirname(p);for(;;){const s=fs.lstatSync(q,{throwIfNoEntry:false});if(s&&(s.isSymbolicLink()||!s.isDirectory()))reject('QA_OUTPUT_ANCESTOR');if(s&&fs.realpathSync(q)!==q)reject('QA_OUTPUT_ANCESTOR');if(q===path.dirname(q))break;q=path.dirname(q);}}
 function writePath(value){const p=normalized(value);if(!owned.some(r=>under(p,r)))reject('QA_OUTPUT_SCOPE');ancestors(p);const s=fs.lstatSync(p,{throwIfNoEntry:false});if(s&&(s.isSymbolicLink()||s.uid!==process.getuid()||s.isFile()&&s.nlink!==1))reject('QA_OUTPUT_BINDING');return p;}
 function readPath(value){const p=normalized(value);if(!owned.some(r=>under(p,r))&&!reads.has(p)&&!dirs.has(p))reject('QA_READ_SCOPE');ancestors(p);return p;}
 function record(op,p){effects.push({op,path:p,utc:new Date(clock()).toISOString()});}
 function fdCheck(fd,writing=true){const x=fds.get(fd);if(!x||writing&&!x.writable)reject('QA_OUTPUT_FD');if(writing)writePath(x.path);else readPath(x.path);const a=fs.fstatSync(fd),b=fs.lstatSync(x.path);if(a.dev!==x.dev||a.ino!==x.ino||b.dev!==a.dev||b.ino!==a.ino||writing&&a.nlink!==1)reject('QA_OUTPUT_FD');return x;}
 const api={constants:fs.constants,
  readFileSync(p,...a){if(typeof p==='number'){fdCheck(p,false);return fs.readFileSync(p,...a);}return fs.readFileSync(readPath(p),...a);},
  readdirSync(p,...a){const n=readPath(p);if(!owned.some(r=>under(n,r))&&!dirs.has(n))reject('QA_READ_DIRECTORY');return fs.readdirSync(n,...a);},
  lstatSync(p,...a){return fs.lstatSync(readPath(p),...a);},statSync(p,...a){return fs.statSync(readPath(p),...a);},realpathSync(p,...a){return fs.realpathSync(readPath(p),...a);},existsSync(p){return fs.existsSync(readPath(p));},
  openSync(p,flags,mode){const writable=typeof flags==='number'?(flags&(fs.constants.O_WRONLY|fs.constants.O_RDWR|fs.constants.O_CREAT|fs.constants.O_APPEND|fs.constants.O_TRUNC))!==0:typeof flags==='string'&&/[wa+]/.test(flags);const n=writable?writePath(p):readPath(p);const fd=fs.openSync(n,typeof flags==='number'?flags|fs.constants.O_NOFOLLOW:flags,mode);const s=fs.fstatSync(fd);fds.set(fd,{path:n,dev:s.dev,ino:s.ino,writable});if(writable)record('open',n);return fd;},
  fstatSync(fd,...a){fdCheck(fd,false);return fs.fstatSync(fd,...a);},readSync(fd,...a){fdCheck(fd,false);return fs.readSync(fd,...a);},
  closeSync(fd){const x=fds.get(fd);if(!x)reject('QA_OUTPUT_FD');try{return fs.closeSync(fd);}finally{fds.delete(fd);}},
  writeSync(fd,...a){const x=fdCheck(fd);record('write',x.path);return fs.writeSync(fd,...a);},fsyncSync(fd){const x=fdCheck(fd);record('fsync',x.path);return fs.fsyncSync(fd);},ftruncateSync(fd,...a){const x=fdCheck(fd);record('truncate',x.path);return fs.ftruncateSync(fd,...a);},
  writeFileSync(p,data,options){if(typeof p==='number'){const x=fdCheck(p);record('writeFileFD',x.path);return fs.writeFileSync(p,data,options);}const n=writePath(p);record('writeFile',n);return fs.writeFileSync(n,data,options);},appendFileSync(p,data,options){const n=writePath(p);record('append',n);return fs.appendFileSync(n,data,options);},
  mkdirSync(p,options){const n=writePath(p);record('mkdir',n);return fs.mkdirSync(n,options);},mkdtempSync(p,options){writePath(p);const result=fs.mkdtempSync(p,options);writePath(result);record('mkdtemp',result);return result;},
  unlinkSync(p){const n=writePath(p);record('unlink',n);return fs.unlinkSync(n);},renameSync(from,to){const a=writePath(from),b=writePath(to);record('rename',b);return fs.renameSync(a,b);},
  copyFileSync(from,to,flags){const a=readPath(from),b=writePath(to);record('copy',b);return fs.copyFileSync(a,b,flags);},
  symlinkSync(target,p,type){const n=writePath(p);normalized(target);if(!owned.some(r=>under(target,r))&&!reads.has(target)&&!dirs.has(target))reject('QA_LINK_TARGET');record('symlink',n);return fs.symlinkSync(target,n,type);},
  linkSync(from,to){const a=writePath(from),b=writePath(to);record('link',b);return fs.linkSync(a,b);},
 };
 // Unsupported promisified/callback/recursive interfaces do not expose raw filesystem capability.
 Object.freeze(api);return Object.freeze({fs:api,writePath,readPath,fdCheck,effects,unsupported:'DEFAULT_DENY; callbacks/promises/cp/rm/native process not yet bridged',closeOwned(){for(const fd of [...fds.keys()])api.closeSync(fd);}});
}
export function boundedBuild(policy,build){return async options=>{if(!options||Object.getPrototypeOf(options)!==Object.prototype||options.plugins?.length||options.stdin===undefined&&(!Array.isArray(options.entryPoints)||options.entryPoints.some(p=>typeof p!=='string'))||options.write!==false&&typeof options.outfile!=='string'||options.outdir!==undefined)reject('QA_BUILD_ROUTE');for(const p of options.entryPoints??[])policy.readPath(p);if(options.outfile!==undefined)policy.writePath(options.outfile);if(options.stdin?.resolveDir)policy.readPath(options.stdin.resolveDir);const allowed=new Set(['entryPoints','stdin','outfile','write','bundle','platform','format','packages','logLevel','metafile','loader','banner','sourcemap','target']);if(Object.keys(options).some(k=>!allowed.has(k)))reject('QA_BUILD_OPTION');const result=await build(options);if(options.outfile!==undefined)policy.writePath(options.outfile);return result;};}
