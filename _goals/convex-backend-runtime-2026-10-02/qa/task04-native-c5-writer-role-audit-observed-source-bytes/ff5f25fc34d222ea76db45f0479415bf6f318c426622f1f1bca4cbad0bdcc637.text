import fs from 'node:fs';import path from 'node:path';import crypto from 'node:crypto';import zlib from 'node:zlib';import {heldSnapshot,heldBytes,actualRow,captureRows} from './preimage-store.mjs';
const raw={...fs},H=b=>crypto.createHash('sha256').update(b).digest('hex');
/** Exact declared metadata + actual generated regular reads in already permitted owned writer roots. No new read route. */
export function inputLedger({declaredPaths,roots,immutablePaths,archiveRoot,recordRoot,historicalRows=[]}){
 const historical=new Map(historicalRows.map(r=>[r.path,r])),declared=new Set([...declaredPaths,...historicalRows.map(r=>r.path)]),immutable=new Set(immutablePaths),events=[],historicalVersions=new Map(),ledgerPath=path.join(recordRoot,'consumed-version-ledger.jsonl'),limit=declaredPaths.length*12000+immutablePaths.length*32+historicalRows.length*4;let ordinal=0;
 const eligible=p=>typeof p==='string'&&(historical.has(p)||!immutable.has(p)&&(declared.has(p)||roots.some(r=>p===r||p.startsWith(r+'/'))))&&!p.startsWith(archiveRoot+'/')&&p!==ledgerPath;
 function append(event){let primary,fd;try{fd=raw.openSync(ledgerPath,raw.constants.O_WRONLY|raw.constants.O_APPEND|raw.constants.O_CREAT|raw.constants.O_NOFOLLOW,0o600);const s=raw.fstatSync(fd),l=raw.lstatSync(ledgerPath);if(!s.isFile()||s.nlink!==1||s.dev!==l.dev||s.ino!==l.ino)throw Error('QA_BRIDGE_LEDGER_BINDING');const b=Buffer.from(JSON.stringify(event)+'\n');let at=0;while(at<b.length){const n=raw.writeSync(fd,b,at,b.length-at);if(!n)throw Error('QA_BRIDGE_LEDGER_WRITE');at+=n;}raw.fsyncSync(fd);}catch(e){primary=e;}if(fd!==undefined)try{raw.closeSync(fd);}catch(e){if(primary)Object.defineProperty(primary,'qaLedgerCloseFailure',{value:e});else primary=e;}if(primary)throw primary;}
 function capture(p,reason){if(!eligible(p))return null;if(events.length>=limit)throw Error('QA_BRIDGE_LEDGER_LIMIT');const s=heldSnapshot(p),row={path:p,bytes:s.bytes.length,sha256:H(s.bytes),identity:s.identity,...(historical.has(p)?{kind:historical.get(p).kind}:{})},captured=captureRows([row],{root:archiveRoot}),event={ordinal:++ordinal,reason,utcBeforeConsumption:new Date().toISOString(),...captured.files[0],classification:declared.has(p)?'DECLARED_METADATA_VERSION':'GENERATED_OWNED_INPUT_VERSION'};if(historical.has(p)){const pinned=historicalVersions.get(p);if(pinned&&(pinned.bytes!==row.bytes||pinned.sha256!==row.sha256))throw Error('QA_BRIDGE_HISTORICAL_VERSION_DRIFT');historicalVersions.set(p,row);const expected=historical.get(p);event.declaredHistorical={bytes:expected.bytes,sha256:expected.sha256};event.declaredHistoricalMatchesActual=expected.bytes===row.bytes&&expected.sha256===row.sha256;event.classification='READONLY_SOURCE_BOUND_HISTORICAL_CURRENT_VERSION';}append(event);events.push(event);return {bytes:s.bytes,event};}
 function verify(event){const current=actualRow(event.path);if(current.sha256!==event.sha256||Object.keys(event.identity).some(k=>current.identity[k]!==event.identity[k]))throw Error('QA_BRIDGE_CONSUMED_VERSION_DRIFT');}
 function readFile(p,options){const c=capture(p,'READ_FILE_BEFORE_RETURN');if(!c)return undefined;const encoding=typeof options==='string'?options:options?.encoding;if(options&&typeof options==='object'&&(options.signal?.aborted))throw options.signal.reason??Error('ABORT_ERR');return {handled:true,value:encoding?c.bytes.toString(encoding):c.bytes};}
 function beforeFDRead(p){return capture(p,'FD_READ_BEFORE_DISPATCH')?.event;}
 function snapshot(reason){const rows=[];for(const p of declared){if(!raw.existsSync(p)){if(historical.has(p))throw Error('QA_BRIDGE_HISTORY_INPUT_MISSING');continue;}const c=capture(p,reason);rows.push(c.event);}return rows;}
 function previous(){for(const e of events){const encoded=heldBytes(e.beforeByteBacking.path),b=e.beforeByteBacking.encoding==='RAW_EXACT_PREIMAGE'?encoded:zlib.gunzipSync(encoded);if(b.length!==e.bytes||H(b)!==e.sha256)throw Error('QA_BRIDGE_LEDGER_ARCHIVE_DRIFT');}return events.map(e=>({...e,role:'CAPTURED_VERSION_BEFORE_ACTUAL_CONSUMPTION_OR_EXPLICIT_SNAPSHOT'}));}
 return Object.freeze({eligible,readFile,beforeFDRead,verify,snapshot,previous,events,ledgerPath,limit});
}
