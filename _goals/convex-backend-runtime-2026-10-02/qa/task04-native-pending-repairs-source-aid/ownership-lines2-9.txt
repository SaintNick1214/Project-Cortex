import {readFileSync,readdirSync,openSync,closeSync,unlinkSync} from 'node:fs';import {resolve} from 'node:path';import {scratch,root,json,sha256,exclusive} from './guard.mjs';
export function track(refs,kind,operation){if(!['created','deleted'].includes(kind)||!Array.isArray(refs)||refs.length>320)throw Error('QA_OWNERSHIP');const budget=json(resolve(root,'cases.json')).budgets.cumulativeRows;
 for(const ref of refs){if(!ref||typeof ref.table!=='string'||!/^[a-zA-Z][a-zA-Z0-9]*$/.test(ref.table)||typeof ref.id!=='string'||ref.id.length>128)throw Error('QA_OWNERSHIP_REF');const digest=sha256(ref.table+':'+ref.id),path=resolve(scratch,'owned-'+kind+'-'+digest+'.json');try{exclusive(path,{operation,kind,table:ref.table,id:ref.id,observedAt:new Date().toISOString()});}catch(error){if(error.code!=='EEXIST')throw error;const old=json(path,scratch,true);if(old.operation!==operation||old.table!==ref.table||old.id!==ref.id)throw Error('QA_OWNERSHIP_COLLISION');}
 const count=readdirSync(scratch).filter(name=>name.startsWith('owned-'+kind+'-')).length;if(count>budget)throw Error('QA_CUMULATIVE_ROW_BUDGET');}
}
export function ownedSnapshot(rows,runId){const operationIds=new Set(rows.filter(x=>x.table==='runtimeModelOperations'&&x.row.tenantId?.startsWith(runId+'-')).map(x=>x.id));return rows.filter(x=>x.row.tenantId?.startsWith(runId+'-')||x.row.subject?.startsWith(runId+'-')||x.row.run===runId||['qa-task04-synthetic-v1','qa-task04-synthetic-embedding-v1'].includes(x.row.root?.manifestId)||x.table==='runtimeModelAttempts'&&operationIds.has(x.row.operationDocumentId)).map(x=>({table:x.table,id:x.id}));}

export function ownershipCounts(){const names=readdirSync(scratch);return {cumulativeCreatedRows:names.filter(n=>n.startsWith('owned-created-')).length,cumulativeDeletedRows:names.filter(n=>n.startsWith('owned-deleted-')).length};}
