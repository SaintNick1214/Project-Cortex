function creationTicketPath(ticket){if(!ticket||!['native','peer-effect','peer-other'].includes(ticket.kind)||!Number.isSafeInteger(ticket.sequence)||ticket.sequence<1)throw Error('QA_CREATION_TICKET');return resolve(scratch,`ticket-${ticket.kind}-${String(ticket.sequence).padStart(4,'0')}.json`);}
function creationState(ticket){const path=creationTicketPath(ticket),data=readFileSync(canonical(path,scratch,true),'utf8');if(Buffer.byteLength(data)>1024)throw Error('QA_CREATION_BYTES');const end=data.indexOf('\n}\n');if(end<0)throw Error('QA_CREATION_PREFIX');const prefix=data.slice(0,end+3),record=JSON.parse(prefix);if(Object.keys(record).sort().join(',')!=='kind,observedAt,sequence'||record.kind!==ticket.kind||record.sequence!==ticket.sequence||!Number.isFinite(Date.parse(record.observedAt)))throw Error('QA_CREATION_PREFIX');const suffix=data.slice(end+3);if(suffix==='')return 0;if(suffix===creationPendingMarker)return 1;if(suffix===creationPendingMarker+creationSettledMarker)return 2;if(suffix===creationPendingMarker+creationResponseMarker)return 3;if(suffix===creationPendingMarker+creationResponseMarker+creationSettledMarker)return 2;throw Error('QA_CREATION_STATE');}
