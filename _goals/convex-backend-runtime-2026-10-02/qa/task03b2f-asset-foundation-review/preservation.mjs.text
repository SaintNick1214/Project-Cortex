import fs from 'node:fs';import ts from 'typescript';
const qa='_goals/convex-backend-runtime-2026-10-02/qa/task03b2f-asset-foundation';
function handlers(file){const ast=ts.createSourceFile(file,fs.readFileSync(file,'utf8'),ts.ScriptTarget.Latest,true);const out=new Map();for(const s of ast.statements)if(ts.isVariableStatement(s))for(const d of s.declarationList.declarations){const c=d.initializer;if(!ts.isCallExpression(c)||!ts.isObjectLiteralExpression(c.arguments[0]))continue;const p=c.arguments[0].properties.find(p=>p.name?.getText(ast)==='handler');if(p&&ts.isPropertyAssignment(p)&&ts.isArrowFunction(p.initializer)&&ts.isBlock(p.initializer.body))out.set(d.name.getText(ast),p.initializer.body.statements.map(n=>n.getText(ast)));}return out;}
const ledger=JSON.parse(fs.readFileSync(qa+'/asset-registration-ledger.json','utf8')).rows;const out=[];for(const module of ['artifacts','attachments']){const before=handlers(qa+'/original/'+module+'.ts'),after=handlers('convex-dev/'+module+'.ts');for(const row of ledger.filter(r=>r.path.startsWith(module+':'))){const name=row.path.split(':')[1],old=before.get(name),now=after.get(name);const guard=now[0];let retained=JSON.stringify(old)===JSON.stringify(now.slice(1));if(module==='attachments'&&name==='get')retained=old.length===1&&old[0]==='return await lookupAttachment(ctx, args.attachmentId, args.tenantId);'&&now[1]==='const attachment: Doc<"attachments"> | null = await lookupAttachment(ctx, args.attachmentId, args.tenantId);'&&now[2]==='return attachment;';out.push({path:row.path,guardFirst:/^await (internalAssetUnavailable|runtimeAssetUnavailable)\(/.test(guard),legacyStatementsRetained:retained});}}
fs.writeFileSync(qa+'-review/body-preservation.json',JSON.stringify(out,null,2));console.log(out);if(out.length!==17||out.some(r=>!r.guardFirst||!r.legacyStatementsRetained))process.exit(1);
