import fs from 'node:fs';
import assert from 'node:assert/strict';

for(const mode of ['no-read','owner-denied','admitted'])for(const c of cases){
 const f=fixture(['write'],'space');
 if(mode!=='no-read')f.db.add('runtimeAuthGrants',{principalId:f.principal._id,membershipId:f.membership._id,tenantId:'tenant-a',tenantEpoch:1,capabilities:['read'],resourceAccess:mode==='owner-denied'?'own':'tenant',version:1});
 const row=f.row({ownerPrincipalId:'other-canonical-owner',streamingState:c.state,content:'unread fixture content',metadata:{config:'unread fixture configuration'},streamingMetadata:{...(c.branch==='missing-session'?{}:{sessionId:'unread fixture session'}),startedAt:1,bytesReceived:7,errorMessage:'unread fixture error'}});

 if(mode==='admitted')assert.equal(readOutput.content,row.content);else assert.equal(readDenied.code,'FORBIDDEN');

 assert.equal(f.db.attempts.length,0);assert.equal(f.db.writes.length,0);
 results.push({handler:c.name,branch:c.branch,readMode:mode,typedCode:thrown.data.code,exactSafeOutput:true,resourceReadControlVerified:true,attempts:0,committed:0,matched:true});
}
fs.writeFileSync(output+'/private-error-admission-results.json',JSON.stringify(results,null,2)+'\n');
