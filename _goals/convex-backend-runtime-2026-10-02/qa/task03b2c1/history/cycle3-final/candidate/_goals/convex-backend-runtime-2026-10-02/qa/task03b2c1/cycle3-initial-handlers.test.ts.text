for (const name of selected) {

      const f = happyCase(name); f.identity(null);

        case "create": expect(row).toMatchObject({ artifactId: "created-a", content: "created content", ownerPrincipalId: f.principal._id, userId: "trusted-user" }); expect(result.versionHistory[0].changedBy).toBe(f.principal._id); break;

        case "setStreamingState": case "startStreaming": case "resumeStreaming": expect(row.streamingState).toBe("streaming"); expect(result.currentState).toBe("streaming"); break;
        case "purgeVersions": expect(row.versionHistory.map((entry: { version: number }) => entry.version)).toEqual([1, 2]); expect(result.versionsRemaining).toBe(2); break;
        case "get": expect(result.artifactId).toBe("artifact-a"); expect(result.versionHistory.length).toBe(2); break;
        case "getByConversation": expect(result.map((entry: { artifactId: string }) => entry.artifactId)).toEqual(["artifact-a"]); expect(JSON.stringify(result)).not.toContain("Private canonical message"); break;
        case "list": expect(result.total).toBe(1); expect(result.artifacts[0].artifactId).toBe("artifact-a"); break;
        case "count": expect(result).toBe(1); break;
        case "getVersion": expect(result).toMatchObject({ version: 1, content: "first", isCurrent: false }); break;
        case "getHistory": expect(result.history.length).toBe(2); expect(result.currentVersion).toBe(2); break;
        case "appendContent": expect(row.content).toBe("first 😀"); expect(result.totalBytesReceived).toBe(5); break;
        case "pauseStreaming": expect(row.streamingState).toBe("paused"); expect(result.contentPreserved).toBe(true); break;
        case "cancelStreaming": expect(row.streamingState).toBe("draft"); expect(row.content).toBe("first"); break;
        case "finalizeStreaming": expect(row.streamingState).toBe("final"); expect(row.versionHistory[2].content).toBe("first"); expect(result.versionCreated).toBe(true); break;
        case "setStreamingError": expect(row.streamingState).toBe("error"); expect(row.streamingMetadata.errorCode).toBe("manual-error"); expect(row.content).toBe("first"); break;

    });
  });

describe("cycle2 retained create admission and immutable repeated sources", () => {
  test.each(["same-grant-capability", "distinct-grant-revocation"] as const)("create %s after insert rejects with reached injection and rollback", async (change) => {
    const f = fixture(change === "same-grant-capability" ? ["read", "write"] : ["write"]);
    const readGrant = change === "same-grant-capability" ? f.grant : f.db.add("runtimeAuthGrants", {
      principalId: f.principal._id, membershipId: f.membership._id, tenantId: "tenant-a", tenantEpoch: 1,
      capabilities: ["read"], resourceAccess: "own", version: 1,
    });
    let injections = 0;
    f.db.hook = (event) => { if (event.action === "insert" && event.table === "artifacts") {
      injections += 1;
      if (change === "same-grant-capability") readGrant.capabilities = ["write"];
      else readGrant.revokedAt = 1;
    } };
    await expect(f.invoke("create", { artifactId: "created", memorySpaceId: "space-a", content: "rollback" }))
      .rejects.toMatchObject({ data: { code: "FORBIDDEN" } });
    expect(injections).toBe(1); expect(f.db.attempts).toHaveLength(1);
    expect(f.db.attempts[0]).toMatchObject({ action: "insert", table: "artifacts" });
    expect(f.db.rows("artifacts")).toEqual([]); expect(f.db.writes).toEqual([]);
  });
  test("create initially absent READ remains a safe receipt after later READ availability", async () => {
    const f = fixture(["write"]); let injections = 0;
    f.db.hook = (event) => { if (event.action === "insert" && event.table === "artifacts") { injections += 1; f.grant.capabilities.push("read"); } };

describe("official Convex native payload witnesses", () => {
  const nativeMetadata = () => ({ integer: BigInt("9223372036854775807"), minimum: BigInt("-9223372036854775808"),
    bytes: new Uint8Array([0, 128, 255]).buffer, nan: Number.NaN, positiveInfinity: Number.POSITIVE_INFINITY,
    negativeInfinity: Number.NEGATIVE_INFINITY, negativeZero: -0 });
  const assertNativeMetadata = (metadata: ReturnType<typeof nativeMetadata>) => {
    expect(metadata.integer).toBe(BigInt("9223372036854775807"));
    expect(metadata.minimum).toBe(BigInt("-9223372036854775808"));
    expect(metadata.bytes).toBeInstanceOf(ArrayBuffer);
    expect([...new Uint8Array(metadata.bytes)]).toEqual([0, 128, 255]);
    expect(Object.is(metadata.nan, Number.NaN)).toBe(true);
    expect(metadata.positiveInfinity).toBe(Number.POSITIVE_INFINITY);
    expect(metadata.negativeInfinity).toBe(Number.NEGATIVE_INFINITY);
    expect(Object.is(metadata.negativeZero, -0)).toBe(true);
  };
  test("native int64, bytes and special floats round trip through registered create/get/update with preserved history", async () => {
    const f = fixture();
    const created = await f.invoke("create", { artifactId: "native", memorySpaceId: "space-a", content: "first", metadata: nativeMetadata() });
    assertNativeMetadata(created.metadata);
    assertNativeMetadata((await f.invoke("get", { artifactId: "native" })).metadata);
    const updated = await f.invoke("update", { artifactId: "native", content: "second" });
    assertNativeMetadata(updated.metadata);
    expect(updated.versionHistory.map((entry: { content: string }) => entry.content)).toEqual(["first", "second"]);
    expect(f.db.attempts.map((event) => event.action)).toEqual(["insert", "patch"]);
  });
  const nativeChanges = [
    { label: "bytes", before: () => new Uint8Array([1, 2]).buffer, after: () => new Uint8Array([1, 3]).buffer },
    { label: "int64", before: () => BigInt("9223372036854775806"), after: () => BigInt("9223372036854775807") },
    { label: "NaN-to-Infinity", before: () => Number.NaN, after: () => Number.POSITIVE_INFINITY },
    { label: "positive-to-negative-Infinity", before: () => Number.POSITIVE_INFINITY, after: () => Number.NEGATIVE_INFINITY },
    { label: "negative-to-positive-zero", before: () => -0, after: () => 0 },
  ];
  for (const kind of ["conversation", "memory"] as const) {
    test(`${kind} native payload and object-key reorder preserves first source witness`, async () => {
      const f = fixture(); const metadata = { z: nativeMetadata(), a: "source" };
      const source = kind === "conversation" ? f.conversation({ metadata, messages: [
        { id: "message-a", role: "user", content: "private source", timestamp: 1, metadata },
      ] }) : f.db.add("memories", { memoryId: "memory-a", tenantId: "tenant-a", memorySpaceId: "space-a",
        ownerPrincipalId: f.principal._id, content: "private source", version: 1, metadata });
      const reference = kind === "conversation" ? { conversationRef: { conversationId: "conversation-a", messageId: "message-a" } }
        : { memoryRefs: [{ memoryId: "memory-a" }] };
      f.row({ artifactId: "first", ...reference }); f.row({ artifactId: "second", ...reference });
      const table = kind === "conversation" ? "conversations" : "memories";
      let queries = 0; let injections = 0;
      f.db.hook = (event) => { if (event.action === "query" && event.table === table && ++queries === 2) {
        injections += 1; source.metadata = { a: "source", z: nativeMetadata() };
        if (kind === "conversation") source.messages[0].metadata = { a: "source", z: nativeMetadata() };
      } };
      const result = await f.invoke("list", {});
      expect(result.total).toBe(2); expect(queries).toBe(2); expect(injections).toBe(1);
      expect(result.artifacts.every((row: { metadata?: unknown }) => row.metadata === undefined)).toBe(true);
      expect(f.db.attempts).toEqual([]);
    });
    test.each(nativeChanges)(`${kind} repeated native $label change denies without delivery or effects`, async ({ before, after }) => {
      const f = fixture();
      const source = kind === "conversation" ? f.conversation({ messages: [
        { id: "message-a", role: "user", content: "private source", timestamp: 1, metadata: { value: before() } },
      ] }) : f.db.add("memories", { memoryId: "memory-a", tenantId: "tenant-a", memorySpaceId: "space-a",
        ownerPrincipalId: f.principal._id, content: "private source", version: 1, metadata: { value: before() } });
      const reference = kind === "conversation" ? { conversationRef: { conversationId: "conversation-a", messageId: "message-a" } }
        : { memoryRefs: [{ memoryId: "memory-a" }] };
      f.row({ artifactId: "first", ...reference }); f.row({ artifactId: "second", ...reference });
      const table = kind === "conversation" ? "conversations" : "memories";
      let queries = 0; let injections = 0; let deliveries = 0;
      f.db.hook = (event) => { if (event.action === "query" && event.table === table && ++queries === 2) {
        injections += 1;
        if (kind === "conversation") source.messages[0].metadata.value = after(); else source.metadata.value = after();
      } };
      await expect(f.invoke("list", {}).then((result) => { deliveries += 1; return result; }))
        .rejects.toMatchObject({ data: { code: "FORBIDDEN" } });
      expect(queries).toBe(2); expect(injections).toBe(1); expect(deliveries).toBe(0);
      expect(f.db.attempts).toEqual([]); expect(f.db.writes).toEqual([]);
    });
  }
  test.each(nativeChanges)("artifact post-effect native $label change denies and rolls back", async ({ before, after }) => {
    const f = fixture(); f.row({ metadata: { value: before() } }); let injections = 0;
    f.db.hook = (event) => { if (event.action === "patch" && event.table === "artifacts") {
      injections += 1; f.db.rows("artifacts")[0].metadata.value = after();
    } };
    await expect(f.invoke("update", { artifactId: "artifact-a", content: "rollback" })).rejects.toMatchObject({ data: { code: "FORBIDDEN" } });
    expect(injections).toBe(1); expect(f.db.attempts).toHaveLength(1);
    expect(f.db.rows("artifacts")[0].content).toBe("first"); expect(f.db.writes).toEqual([]);
    // The before-value is compared with the official codec by the handler; the
    // transaction fixture also restores the original native value on failure.
    const restored = f.db.rows("artifacts")[0].metadata.value;
    if (before() instanceof ArrayBuffer) expect([...new Uint8Array(restored)]).toEqual([1, 2]);
    else expect(Object.is(restored, before())).toBe(true);
  });
  test.each(["create", "update"] as const)("%s native codec errors are sanitized and precede effects", async (name) => {
    const f = fixture(); if (name === "update") f.row();
    const operation = f.invoke(name, { artifactId: "artifact-a", memorySpaceId: "space-a", content: "manual",
      metadata: { privateValue: "never expose this metadata", outOfNativeRange: BigInt("18446744073709551616") } });
    await expect(operation).rejects.toMatchObject({ data: { code: "INVALID_ARGUMENT", message: "Artifact operation denied." } });
    await operation.catch((error) => {
      expect(String(error)).not.toContain("never expose this metadata"); expect(String(error)).not.toContain("18446744073709551616");
    });
    expect(f.db.attempts).toEqual([]); expect(f.db.writes).toEqual([]);
  });
