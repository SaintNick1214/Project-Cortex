import { describe, expect, test, jest } from "@jest/globals";
import { fixture, happyCase, native, selected } from "./fixture";

for (const name of selected) {
  describe(`native artifacts:${name}`, () => {
    test("requires verified identity before artifact hydration or effects", async () => {
      const f = happyCase(name); f.identity(null);
      await expect(f.invoke(name, f.args)).rejects.toMatchObject({ data: { code: "UNAUTHENTICATED" } });
      expect(f.db.reads.filter((event) => event.table === "artifacts")).toEqual([]);
      expect(f.db.writes).toEqual([]);
      expect(native[name].isPublic).toBe(true);
    });
    test("preserves the admitted text operation's actual outcome", async () => {
      const f = happyCase(name); const result = await f.invoke(name, f.args);
      const row = f.db.rows("artifacts")[0];
      switch (name) {
        case "create": expect(row).toMatchObject({ artifactId: "created-a", content: "created content", ownerPrincipalId: f.principal._id, userId: "trusted-user" }); expect(result.versionHistory[0].changedBy).toBe(f.principal._id); break;
        case "update": expect(row.content).toBe("updated content"); expect(row.version).toBe(3); expect(result.versionHistory[2].changedBy).toBe(f.principal._id); break;
        case "deleteArtifact": expect(row.isDeleted).toBe(true); expect(row.tombstonedAt).toBeGreaterThan(0); expect(f.db.rows("runtimeAuthTombstones")[0].resourceId).toBe("artifact-a"); expect(result.restorable).toBe(false); break;
        case "undo": expect(row.versionPointer).toBe(1); expect(row.content).toBe("first"); expect(result.canRedo).toBe(true); break;
        case "redo": expect(row.versionPointer).toBe(2); expect(row.content).toBe("second"); expect(result.canUndo).toBe(true); break;
        case "setStreamingState": case "startStreaming": case "resumeStreaming": expect(row.streamingState).toBe("streaming"); expect(result.currentState).toBe("streaming"); break;
        case "purgeVersions": expect(row.versionHistory.map((entry: { version: number }) => entry.version)).toEqual([1, 2]); expect(result.versionsRemaining).toBe(2); break;
        case "get": expect(result.artifactId).toBe("artifact-a"); expect(result.versionHistory.length).toBe(2); break;
        case "getByConversation": expect(result.map((entry: { artifactId: string }) => entry.artifactId)).toEqual(["artifact-a"]); expect(JSON.stringify(result)).not.toContain("Private canonical message"); break;
        case "list": expect(result.total).toBe(1); expect(result.artifacts[0].artifactId).toBe("artifact-a"); break;
        case "count": expect(result).toBe(1); break;
        case "getVersion": expect(result).toMatchObject({ version: 1, content: "first", isCurrent: false }); break;
        case "getHistory": expect(result.history.length).toBe(2); expect(result.currentVersion).toBe(2); break;
        case "appendContent": expect(row.content).toBe("first 😀"); expect(result.totalBytesReceived).toBe(5); break;
        case "pauseStreaming": expect(row.streamingState).toBe("paused"); expect(result.contentPreserved).toBe(true); break;
        case "cancelStreaming": expect(row.streamingState).toBe("draft"); expect(row.content).toBe("first"); break;
        case "finalizeStreaming": expect(row.streamingState).toBe("final"); expect(row.versionHistory[2].content).toBe("first"); expect(result.versionCreated).toBe(true); break;
        case "setStreamingError": expect(row.streamingState).toBe("error"); expect(row.streamingMetadata.errorCode).toBe("manual-error"); expect(row.content).toBe("first"); break;
        case "retryFromError": expect(row.streamingState).toBe("draft"); expect(row.content).toBe("first"); expect(result.contentCleared).toBe(false); break;
      }
    });
  });
}

  });
  test("expiry after the last control await blocks delivery", async () => {
    const measured = fixture(); measured.row(); await measured.invoke("get", { artifactId: "artifact-a" });
    const finalReadCount = measured.db.reads.filter((event) => event.table === "runtimeAuthTombstones").length;
    expect(finalReadCount).toBeGreaterThan(0);
    const f = fixture(); f.row(); const now = Date.now(); f.grant.expiresAt = now + 500;
    const clock = jest.spyOn(Date, "now").mockReturnValue(now);
    let calls = 0;
    f.db.hook = (event) => { if (event.table === "runtimeAuthTombstones" && ++calls === finalReadCount) clock.mockReturnValue(now + 1000); };
    try { await expect(f.invoke("get", { artifactId: "artifact-a" })).rejects.toMatchObject({ data: { code: "FORBIDDEN" } }); }
    finally { clock.mockRestore(); }
    expect(calls).toBe(finalReadCount);
    expect(f.db.reads[f.db.reads.length - 1]).toMatchObject({ table: "runtimeAuthTombstones", filters: [
      ["tenantId", "tenant-a"], ["memorySpaceId", undefined], ["resourceType", "artifact"], ["resourceId", "artifact-a"]] });

describe("retained admission and transactional source witnesses", () => {
  test("an initially unavailable READ is never newly adopted after a write", async () => {
    const f = fixture(["write"]); f.row();
    f.db.hook = (event) => { if (event.action === "patch") f.grant.capabilities.push("read"); };
    expect(await f.invoke("update", { artifactId: "artifact-a", content: "manual" })).toEqual({ success: true, artifactId: "artifact-a" });
    expect(f.db.rows("artifacts")[0].content).toBe("manual");
  });
  test.each(["principal", "membership", "scope", "resource"] as const)("late %s control failure rolls back the admitted text mutation", async (control) => {
    const f = fixture(); f.row();
    f.db.hook = (event) => { if (event.action !== "patch") return;
      if (control === "principal") f.principal.deletedAt = 1;
      if (control === "membership") f.membership.revokedAt = 1;
      if (control === "scope") f.db.rows("runtimeAuthScopes").find((row) => row.memorySpaceId === "space-a")!.epoch = 2;
      if (control === "resource") f.db.add("runtimeAuthTombstones", { tenantId: "tenant-a", memorySpaceId: "space-a", resourceType: "artifact", resourceId: "artifact-a", deletedAt: 1 });
    };
    await expect(f.invoke("update", { artifactId: "artifact-a", content: "unsafe" })).rejects.toMatchObject({ data: { code: "FORBIDDEN" } });
    expect(f.db.rows("artifacts")[0].content).toBe("first"); expect(f.db.writes).toEqual([]); expect(f.db.attempts).toHaveLength(1);
  });
  test("never adopt changed inserted payload after effects", async () => {
    const f = fixture(); f.db.hook = (event) => { if (event.action === "insert" && event.table === "artifacts") f.db.rows("artifacts")[0].content = "changed after insert"; };
    await expect(f.invoke("create", { memorySpaceId: "space-a", content: "expected" })).rejects.toMatchObject({ data: { code: "FORBIDDEN" } });
    expect(f.db.rows("artifacts")).toEqual([]); expect(f.db.writes).toEqual([]);
  });
  test("linked anchor changed after effect is denied with full rollback", async () => {
    const f = fixture(); const conversation = f.conversation(); f.row({ conversationRef: { conversationId: "conversation-a", messageId: "message-a" } });
    f.db.hook = (event) => { if (event.action === "patch") conversation.messages[0].id = "replacement"; };
    await expect(f.invoke("update", { artifactId: "artifact-a", content: "unsafe" })).rejects.toMatchObject({ data: { code: "FORBIDDEN" } });
    expect(f.db.rows("artifacts")[0].content).toBe("first"); expect(f.db.writes).toEqual([]);
  });
  test("file-bearing current row is rejected by all applicable selected public APIs before effects", async () => {
    for (const name of selected.filter((name) => name !== "create")) {
      const f = happyCase(name); f.db.rows("artifacts")[0].fileRef = { storageId: "_storage:foreign", mimeType: "image/png", size: 1 };
      await expect(f.invoke(name, f.args)).rejects.toMatchObject({ data: { code: "CAPABILITY_NOT_READY" } });
      expect(f.db.attempts).toEqual([]);
    }
  });
  test("canonical memory source owner and lifecycle are required", async () => {
    const f = fixture(); f.row({ memoryRefs: [{ memoryId: "memory-a" }] });
    f.db.add("memories", { memoryId: "memory-a", tenantId: "tenant-a", memorySpaceId: "space-a", ownerPrincipalId: f.principal._id,
      content: "private source", version: 1 });
    expect((await f.invoke("get", { artifactId: "artifact-a" })).artifactId).toBe("artifact-a");
    f.db.rows("memories")[0].tombstonedAt = 1;
    await expect(f.invoke("update", { artifactId: "artifact-a", content: "unsafe" })).rejects.toMatchObject({ data: { code: "FORBIDDEN" } }); expect(f.db.attempts).toEqual([]);
  });
  test("completed streaming keeps version history; cancellation and retry preserve selected content behavior", async () => {
    const f = fixture(); f.row();
    const started = await f.invoke("startStreaming", { artifactId: "artifact-a", streamSource: "ai_generation", metadata: { verifiedTool: true } });
    await f.invoke("appendContent", { artifactId: "artifact-a", sessionId: started.sessionId, chunk: " body" });
    await f.invoke("pauseStreaming", { artifactId: "artifact-a", sessionId: started.sessionId });
    await f.invoke("finalizeStreaming", { artifactId: "artifact-a", sessionId: started.sessionId });
    expect(f.db.rows("artifacts")[0].versionHistory[2]).toMatchObject({ version: 3, content: "first body", changedBy: f.principal._id });
    expect(f.db.rows("artifacts")[0].streamingMetadata.sessionId).toBeUndefined();
    expect(f.db.rows("artifacts")[0].metadata).toBeUndefined();
  });
});
