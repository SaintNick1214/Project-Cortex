import ts from 'typescript';import fs from 'node:fs';import assert from 'node:assert/strict';import {resolve} from 'node:path';
const base='_goals/convex-backend-runtime-2026-10-02/qa/task03-foundation-live-preparation-c3',rows=[],http=[];
for(const name of fs.readdirSync('convex-dev').filter(x=>x.endsWith('.ts'))){const source=ts.createSourceFile(name,fs.readFileSync('convex-dev/'+name,'utf8'),ts.ScriptTarget.Latest,true),builders=new Map();for(const s of source.statements){if(ts.isImportDeclaration(s)&&s.importClause?.namedBindings&&ts.isNamedImports(s.importClause.namedBindings)&&s.moduleSpecifier.text==='./_generated/server')for(const e of s.importClause.namedBindings.elements){const original=e.propertyName?.text??e.name.text;if(/^(internal)?(Query|Mutation|Action)$/.test(original)||['query','mutation','action','httpAction'].includes(original))builders.set(e.name.text,original);}}
for(const s of source.statements){if(!ts.isVariableStatement(s))continue;for(const d of s.declarationList.declarations){if(!d.initializer||!ts.isCallExpression(d.initializer)||!ts.isIdentifier(d.initializer.expression)||!builders.has(d.initializer.expression.text))continue;assert.ok(s.modifiers?.some(m=>m.kind===ts.SyntaxKind.ExportKeyword),'unexported registration');const b=builders.get(d.initializer.expression.text);if(b==='httpAction'){http.push(name+':'+d.name.text);continue;}rows.push({path:name.slice(0,-3)+':'+d.name.text,kind:b.replace(/^internal/,'').toLowerCase(),visibility:b.startsWith('internal')?'internal':'public'});}}
}
const ledger=JSON.parse(fs.readFileSync(base+'/path-ledger.json'));const sort=rs=>rs.sort((a,b)=>a.path.localeCompare(b.path));assert.deepEqual(sort(rows),sort(ledger.cases.map(({path,kind,visibility})=>({path,kind,visibility}))));assert.equal(http.length,0);const result={registered:rows.length,public:rows.filter(x=>x.visibility==='public').length,internal:rows.filter(x=>x.visibility==='internal').length,modules:new Set(rows.map(x=>x.path.split(':')[0])).size,http:0,exactASTLedgerEquality:true};fs.writeFileSync('work/resume/current-guard-preparation-c3-review/ast-inventory.json',JSON.stringify({summary:result,rows},null,2)+'\n');console.log(JSON.stringify(result));
