  11: interface Filters { type?: string; userId?: string; createdAfter?: number; createdBefore?: number; updatedAfter?: number; updatedBefore?: number }
  12: function filter(rows: Doc<"immutable">[], args: Filters) {
  13:   return rows.filter((row) => (args.type === undefined || row.type === args.type) && (args.userId === undefined || row.userId === args.userId)
  14:     && (args.createdAfter === undefined || row.createdAt > args.createdAfter) && (args.createdBefore === undefined || row.createdAt < args.createdBefore)
  15:     && (args.updatedAfter === undefined || row.updatedAt > args.updatedAfter) && (args.updatedBefore === undefined || row.updatedAt < args.updatedBefore));
  16: }
  17: export function versions(row: Doc<"immutable">) {
  18:   return [...row.previousVersions, { version: row.version, data: row.data, timestamp: row.updatedAt, metadata: row.metadata }]
  19:     .map((version) => ({ ...version, type: row.type, id: row.id, userId: row.userId, createdAt: row.createdAt })).sort((a, b) => a.version - b.version);
  20: }

  33:         previousVersions: [...row.previousVersions, { version: row.version, data: row.data, timestamp: row.updatedAt, metadata: row.metadata }],

  40:     const updated = await ctx.db.get("immutable", id); if (!updated) throw new ConvexError("IMMUTABLE_ENTRY_NOT_FOUND");

  42:       () => ({ updated: true as const, type: args.type, id: args.id, tenantId: auth.tenantId, memorySpaceId: auth.memorySpaceId, ...(row ? {} : { createdId: id }) }));

  80: export const purgeMany = mutation({ args: filterArgs, handler: async (ctx, args) => {
  81:   const auth = await authority(ctx, "write", args); const rows = filter(await candidates(ctx, "immutable", auth, "write"), args);
  82:   const reads = await Promise.all(rows.map((row) => readAuthority(ctx, auth, "immutable", row)));

  88: } });
