   9: interface Filters { namespace: string; keyPrefix?: string; userId?: string; updatedBefore?: number; updatedAfter?: number }
  10: function filter(rows: Doc<"mutable">[], args: Filters) { return rows.filter((row) => row.namespace === args.namespace
  11:   && (args.keyPrefix === undefined || row.key.startsWith(args.keyPrefix)) && (args.userId === undefined || row.userId === args.userId)
  12:   && (args.updatedBefore === undefined || row.updatedAt < args.updatedBefore) && (args.updatedAfter === undefined || row.updatedAt > args.updatedAfter)); }
  13: function numeric(value: unknown, amount: unknown, decrement = false): number {
  14:   const delta = amount === undefined ? 1 : amount;
  15:   if (typeof value !== "number" || !Number.isFinite(value) || typeof delta !== "number" || !Number.isFinite(delta)) invalid();
  16:   const result = value + (decrement ? -delta : delta); if (!Number.isFinite(result)) invalid(); return result;
  17: }
  18: function updateValue(value: unknown, operation: string, operand: unknown): unknown {
  19:   if (operation === "increment" || operation === "decrement") return numeric(value, operand, operation === "decrement");
  20:   if (operation === "append") { if (operand === undefined) invalid(); if (!Array.isArray(value)) throw new ConvexError("MUTABLE_VALUE_NOT_ARRAY"); return [...value, operand]; }
  21:   if (operand === undefined) invalid(); return operand;
  22: }

  72: export const transaction = mutation({ args: { ...scopeArgs, operations: v.array(v.object({ op: v.union(v.literal("set"), v.literal("update"), v.literal("delete"), v.literal("increment"), v.literal("decrement")), namespace: v.string(), key: v.string(), value: v.optional(v.any()), amount: v.optional(v.number()) })) }, handler: async (ctx, args) => {
  73:   const auth = await authority(ctx, "write", args);
  74:   const state = new Map<string, { row: Doc<"mutable"> | null; value: unknown; deleted: boolean; read: Awaited<ReturnType<typeof readAuthority>> }>();
  75:   const plans: Array<{ operation: typeof args.operations[number]; value: unknown }> = [];
  76:   // Validate authorization and functional operations for the ENTIRE batch before the first write.
  77:   for (const operation of args.operations) {
  78:     const key = sourceId("mutable", operation.namespace, operation.key);
  79:     let current = state.get(key);
  80:     if (!current) {
  81:       const row = await mutableRow(ctx, auth, operation.namespace, operation.key, "write"); await newResource(ctx, auth, key);
  82:       const read = row ? await readAuthority(ctx, auth, "mutable", row) : await readResourceAuthority(ctx, auth, newTarget(auth, key));
  83:       current = { row, value: row?.value, deleted: false, read }; state.set(key, current);
  84:     }
  85:     if (current.deleted) invalid();
  86:     if (operation.op !== "set" && current.value === undefined) throw new ConvexError("MUTABLE_KEY_NOT_FOUND");
  87:     if (operation.op === "delete") current.deleted = true;
  88:     else if (operation.op === "increment" || operation.op === "decrement") current.value = numeric(current.value, operation.amount, operation.op === "decrement");
  89:     else { if (operation.value === undefined) invalid(); current.value = operation.value; }
  90:     plans.push({ operation, value: current.value });
  91:   }
  92:   const results: Array<Doc<"mutable"> | { deleted: boolean; namespace: string; key: string } | { updated: boolean; namespace: string; key: string }> = [];
  93:   for (const { operation, value } of plans) {
  94:     const key = sourceId("mutable", operation.namespace, operation.key); const current = state.get(key)!;
  95:     const row = await mutableRow(ctx, auth, operation.namespace, operation.key, "write");
  96:     if (operation.op === "delete") { if (!row) throw new ConvexError("MUTABLE_KEY_NOT_FOUND"); await check(ctx, "mutable", row, auth, "write"); await tombstone(ctx, "mutable", row); results.push({ deleted: true, namespace: operation.namespace, key: operation.key }); }
  97:     else {
  98:       const now = Date.now(); if (row) { await check(ctx, "mutable", row, auth, "write"); await ctx.db.patch("mutable", row._id, { value, updatedAt: now }); } else await newResource(ctx, auth, key);
  99:       const id = row?._id ?? await ctx.db.insert("mutable", { namespace: operation.namespace, key: operation.key, value, ...canonicalFields(auth), createdAt: now, updatedAt: now });
 100:       const updated = await ctx.db.get("mutable", id); if (!updated) throw new ConvexError("MUTABLE_KEY_NOT_FOUND");
 101:       await check(ctx, "mutable", updated, auth, "write"); results.push(current.read ? updated : { updated: true, namespace: operation.namespace, key: operation.key }); current.row = updated;
 102:     }
 103:   }
 104:   await finalChecks(ctx, [{ auth, capability: "write" }, ...[...state.values()].flatMap((current) => current.read ? [{ auth: current.read, capability: "read" as const,
 105:     ...(!current.deleted && current.row ? { resource: resource("mutable", current.row) } : {}) }] : [])]);
 106:   return { success: true, operationsExecuted: plans.length, results };
 107: } });
