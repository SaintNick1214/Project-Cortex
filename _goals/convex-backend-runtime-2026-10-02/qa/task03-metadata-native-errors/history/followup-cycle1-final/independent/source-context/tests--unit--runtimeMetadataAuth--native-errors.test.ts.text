  49: function nativeProtocol(f: F) {
  50:   const original = f.db.query.bind(f.db);
  51:   const traces: Array<{ table: string; index?: string; keys: [string, unknown][]; limit: number; ids: string[] }> = [];

  53:   f.db.query = (table) => {
  54:     const query = original(table);
  55:     const index = query.withIndex.bind(query);
  56:     const take = query.take.bind(query);
  57:     const keys: [string, unknown][] = [];
  58:     let indexName: string | undefined;
  59:     // Retain the original database selector as well as the observation.
  60:     query.withIndex = (name, callback) => {
  61:       indexName = name;
  62:       return index(name, (selector) => {
  63:         const observed = { eq: (key: string, value: unknown) => { keys.push([key, value]); selector.eq(key, value); return observed; } };
  64:         return callback(observed);
  65:       });
  66:     };
  67:     query.take = async (limit) => {
  68:       const rows = await take(limit);
  69:       traces.push({ table, index: indexName, keys: [...keys], limit, ids: rows.map((row) => row._id) });
  70:       return rows;
  71:     };

  77:     return query;
  78:   };
  79:   return { traces, uniqueCalls: () => uniqueCalls };
