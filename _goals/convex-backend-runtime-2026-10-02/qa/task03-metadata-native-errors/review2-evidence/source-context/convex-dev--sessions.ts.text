  63: function sameReference(a: RuntimeAuthorityReference, b: RuntimeAuthorityReference): boolean {
  64:   return a.principalId === b.principalId && a.principalVersion === b.principalVersion && a.membershipId === b.membershipId
  65:     && a.membershipVersion === b.membershipVersion && a.grantId === b.grantId && a.grantVersion === b.grantVersion

  67: }
  68: async function backgroundCheck(ctx: MutationCtx, reference: RuntimeAuthorityReference, row: Doc<"sessions">): Promise<RuntimeAuthority> {
  69:   if (!row.authorityReference || !sameReference(row.authorityReference, reference) || row.tombstonedAt !== undefined) denied();
  70:   const auth = await recheckAuthority(ctx, row.authorityReference, { capability: "write", resource: resource("sessions", row) });
  71:   await finalChecks(ctx, [{ auth, capability: "write", resource: resource("sessions", row) }]);
  72:   // Background effects use the session creator's actor, not breadth of a later space grant.
  73:   if (row.ownerPrincipalId !== auth.principalId || row.userId !== auth.userId
  74:     || row.tenantId !== auth.tenantId || row.memorySpaceId !== auth.memorySpaceId) denied();
  75:   return auth;
  76: }
