const results=[];
function setup(capability,defect){
  const f=fixture({space:'space-a',capabilities:[capability],access:'space'});
  f.db.seed('runtimeAuthScopes',{tenantId:'tenant-a',memorySpaceId:'space-b',epoch:1,createdAt:1000});
  f.db.seed('runtimeAuthGrants',{...f.grant,_id:'runtimeAuthGrants:space-b',memorySpaceId:'space-b'});
  f.seedMutable({memorySpaceId:'space-b',_id:'mutable:private-b',value:'private-b-value'});
  if(defect==='scope')f.db.seed('runtimeAuthScopes',{...f.db.table('runtimeAuthScopes').find(r=>r.memorySpaceId==='space-a'),_id:'runtimeAuthScopes:duplicate-a'});
  if(defect==='space-tombstone')for(const suffix of ['one','two'])f.db.seed('runtimeAuthTombstones',{_id:'runtimeAuthTombstones:private-'+suffix,tenantId:'tenant-a',memorySpaceId:'space-a',resourceType:'memorySpace',resourceId:'space-a',deletedAt:1000});
  let reached=0;const old=f.db.query.bind(f.db);
  f.db.query=(table)=>{const q=old(table),take=q.take.bind(q);q.take=async(limit)=>{const rows=await take(limit);if(rows.length>1)reached++;return rows;};q.unique=()=>QueryImpl.prototype.unique.call({take:q.take,tableNameForErrorMessages:table});return q;};
  return {f,reached:()=>reached};
}
async function run(f,operation){try{return {accepted:true,value:await operation(),attempts:f.db.attemptedWrites,writes:f.db.writes}}catch(e){return {accepted:false,errorType:e.constructor.name,errorData:e.data,errorMessage:e.message,attempts:f.db.attemptedWrites,writes:f.db.writes}}}
for(const capability of ['read','write'])for(const defect of ['none','scope','space-tombstone']){
  const {f,reached}=setup(capability,defect);
  const result=await run(f,()=>f.run(capability==='read'?mutable.get:mutable.set,{namespace:'counter',key:'one',...(capability==='write'?{value:'replacement'}:{})}));
  results.push({capability,defect,kind:'actual registered handler omitted selectors',duplicateLookupReached:reached(),...result});
  assert.equal(result.accepted,defect!=='none');
  assert.equal(reached(),defect==='none'?0:1);
  if(result.accepted)assert.equal(result.writes,capability==='write'?1:0);
}
