  11: function handler<Args, Result>(registration: unknown): (ctx: MutationCtx, args: Args) => Promise<Result> {
  12:   return (registration as { _handler: (ctx: MutationCtx, args: Args) => Promise<Result> })._handler;
  13: }

  28: type Row = { _id: string; [field: string]: unknown };
  29: /** In-memory adapter harness, not a substitute for Convex transaction/live endpoint QA in 03B. */
  30: class MemoryDb {
  31:   readonly rows = new Map<string, Row[]>();
  32:   readonly writes: string[] = [];
  33:   private next = 1;
  34: 
  35:   normalizeId(table: string, id: string) { return id.startsWith(`${table}/`) ? id : null; }
  36:   query(table: string) {
  37:     const predicates: Array<[string, unknown]> = [];
  38:     const selected = () => (this.rows.get(table) ?? []).filter((row) => predicates.every(([key, value]) => row[key] === value));
  39:     const result = {
  40:       withIndex: (_name: string, callback: (q: { eq: (field: string, value: unknown) => unknown }) => unknown) => {
  41:         const q = { eq: (field: string, value: unknown) => { predicates.push([field, value]); return q; } };
  42:         callback(q); return result;
  43:       },
  44:       unique: async () => { const values = selected(); if (values.length > 1) throw new Error("Duplicate unique control record"); return values[0] ?? null; },
  45:       take: async (limit: number) => selected().slice(0, limit),
  46:       collect: async () => selected(),
  47:     };
  48:     return result;
  49:   }
  50:   async get(table: string, id: string) { return (this.rows.get(table) ?? []).find((row) => row._id === id) ?? null; }
  51:   async insert(table: string, fields: Record<string, unknown>) {
  52:     const _id = `${table}/${this.next++}`;
  53:     this.rows.set(table, [...(this.rows.get(table) ?? []), { ...fields, _id }]);
  54:     this.writes.push(`insert:${table}`); return _id;
  55:   }
  56:   async patch(table: string, id: string, fields: Record<string, unknown>) {
  57:     const row = await this.get(table, id);
  58:     if (!row) throw new Error("Missing row");
  59:     Object.assign(row, fields); this.writes.push(`patch:${table}`);
  60:   }
  61: }
