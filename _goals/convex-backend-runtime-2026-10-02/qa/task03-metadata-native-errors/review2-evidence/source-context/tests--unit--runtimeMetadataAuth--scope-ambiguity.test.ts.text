  48: function protocol(f: F, beforeTake?: (table: string, keys: [string, unknown][]) => void) {
  49:   const original = f.db.query.bind(f.db);
  50:   const traces: Array<{ table: string; index?: string; keys: [string, unknown][]; limit: number; ids: string[] }> = [];
  51:   let uniqueCalls = 0;
  52:   f.db.query = (table) => {
  53:     const q = original(table); const index = q.withIndex.bind(q); const take = q.take.bind(q);
  54:     const keys: [string, unknown][] = []; let indexName: string | undefined;
  55:     q.withIndex = (name, callback) => {
  56:       indexName = name;
  57:       return index(name, (selector) => {
  58:         const observed = { eq: (key: string, value: unknown) => { keys.push([key, value]); selector.eq(key, value); return observed; } };
  59:         return callback(observed);
  60:       });
  61:     };
  62:     q.take = async (limit) => {
  63:       beforeTake?.(table, keys);
  64:       const rows = await take(limit); traces.push({ table, index: indexName, keys: [...keys], limit, ids: rows.map((row) => row._id) });
  65:       return rows;
  66:     };
  67:     // Keep the installed native diagnostic if a regression reintroduces unique().
  68:     // The fixture's inferred non-null signature does not change native null at runtime.
  69:     q.unique = async () => { uniqueCalls++; return (await QueryImpl.prototype.unique.call({ take: q.take, tableNameForErrorMessages: table }))!; };
  70:     return q;
  71:   };
  72:   return { traces, uniqueCalls: () => uniqueCalls };
  73: }
  74: async function failure(operation: Promise<unknown>): Promise<unknown> {
