const fs=require('node:fs');const assert=require('node:assert/strict');const crypto=require('node:crypto');const ts=require('/workspace/Project-Cortex/node_modules/typescript');const scratch='/tmp/task03c2-a-review1-4cCdIR';
function methods(text){const ast=ts.createSourceFile('module.ts',text,ts.ScriptTarget.Latest,true);const result=new Map();for(const node of ast.statements){if(!ts.isClassDeclaration(node))continue;for(const member of node.members){if(ts.isMethodDeclaration(member))result.set(member.name.getText(ast),member.getText(ast));}}return result;}
const hashes=[];for(const [file,changed] of [['src/users/index.ts',['update','updateMany']],['src/sessions/index.ts',['expireIdle']]]){const before=methods(fs.readFileSync(`${scratch}/task03c2a-browser-baseline/${file}`,'utf8'));const after=methods(fs.readFileSync(`${scratch}/candidate/${file}`,'utf8'));assert.deepEqual([...before.keys()],[...after.keys()]);for(const[name,text]of before){if(changed.includes(name))continue;assert.equal(after.get(name),text);hashes.push({file,method:name,sha256:crypto.createHash('sha256').update(text).digest('hex')});}}
assert.equal(hashes.length,36);const baseRoot=fs.readFileSync(`${scratch}/task03c2a-browser-baseline/src/index.ts`,'utf8');assert.equal(fs.readFileSync(`${scratch}/candidate/src/index.ts`,'utf8'),baseRoot.replace('export { UserValidationError } from "./users";','export { UserValidationError, UserProfileWriteReceiptError } from "./users";').replace('export { SessionValidationError } from "./sessions";','export { SessionValidationError, SessionCapabilityError } from "./sessions";'));fs.writeFileSync(`${scratch}/method-verification.json`,JSON.stringify({unchangedMethodCount:hashes.length,methods:hashes,rootOnlyNecessaryErrorExports:true},null,2)+'\n');console.log(JSON.stringify({unchangedMethodCount:hashes.length,rootOnlyNecessaryErrorExports:true}));
