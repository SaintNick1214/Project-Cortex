/** Reproduce bounded actual-registration closure evidence; frozen 03A catalog stays historical. */
import fs from "node:fs";
import path from "node:path";
import crypto from "node:crypto";
import ts from "typescript";
const root = process.cwd();
const destination = process.argv[2] ?? path.join(root, "_goals/convex-backend-runtime-2026-10-02/qa/task03b1-mf");
const baseline = JSON.parse(fs.readFileSync(path.join(root, "_goals/convex-backend-runtime-2026-10-02/qa/task03a/public-path-inventory.json"), "utf8"));
const frozen = baseline.endpoints.filter(entry => ["convex-dev/memories.ts", "convex-dev/facts.ts"].includes(entry.file));
const entries = [];
for (const file of ["convex-dev/memories.ts", "convex-dev/facts.ts"]) {
  const source = fs.readFileSync(path.join(root, file), "utf8");
  const tree = ts.createSourceFile(file, source, ts.ScriptTarget.Latest, true, ts.ScriptKind.TS);
  let builders = 0;
  function count(node) {
    if (ts.isCallExpression(node) && ts.isIdentifier(node.expression) && /^(query|mutation|action|internalQuery|internalMutation)$/.test(node.expression.text)) builders++;
    ts.forEachChild(node, count);
  }
  count(tree);
  const moduleEntries = [];
  for (const statement of tree.statements) {
    if (!ts.isVariableStatement(statement) || !statement.modifiers?.some(modifier => modifier.kind === ts.SyntaxKind.ExportKeyword)) continue;
    for (const declaration of statement.declarationList.declarations) {
      const call = declaration.initializer;
      if (!ts.isCallExpression(call) || !ts.isIdentifier(call.expression) || !/^(query|mutation|action|internalQuery|internalMutation)$/.test(call.expression.text)) continue;
      const name = declaration.name.getText(tree); const kind = call.expression.text;
      const text = call.getText(tree);
      const guard = kind.startsWith("internal") ? name === "purgeAll" ? "deployment-operator internal invocation" : "recheckDataAuthority"
        : kind === "action" ? "requireActionDataAuthority" : "requireDataAuthority";
      if (name !== "purgeAll" && !text.includes(`${guard}(`)) throw new Error(`Missing actual handler guard: ${file}:${name}`);
      if (!kind.startsWith("internal") && /\.query\(|\.get\(/.test(text)) throw new Error(`Unexpected raw public DB selection: ${file}:${name}`);
      const scopedHelpers = [...new Set([...text.matchAll(/\b((?:get|list|search)Scoped\w+|getHistoricalScopedFact|assertData\w+|manualDataSource|reviseDataSource|tombstoneDataRow|recheckActionDataAuthority|rejectUnqualifiedEmbedding|(?:memory|fact)MutationResult|canReadMutationRows|finalizeMutationRowsRead|prepareMutationRowsWrite|finalizeDataRead|recheckSourceWitnesses)\(/g)].map(match => match[1]))];
      moduleEntries.push({ path: `${path.basename(file, ".ts")}:${name}`, file, line: tree.getLineAndCharacterOfPosition(declaration.getStart(tree)).line + 1,
        kind: kind.replace(/^internal/, "").toLowerCase(), visibility: kind.startsWith("internal") ? "internal" : "public",
        guard, scopedHelpers, baselineDisposition: frozen.find(entry => entry.path === `${path.basename(file, ".ts")}:${name}`)?.disposition,
        semanticBoundary: text.includes("rejectUnqualifiedEmbedding(") ? "PROFILE_NOT_READY for unqualified caller embedding; modern profile wiring Task09/10 pending" : undefined });
    }
  }
  if (builders !== moduleEntries.length) throw new Error(`Unresolved registrations: ${file}`);
  entries.push(...moduleEntries);
}
const actualPaths = entries.map(entry => entry.path).sort();
if (JSON.stringify(actualPaths) !== JSON.stringify(frozen.map(entry => entry.path).sort())) throw new Error("Bounded baseline paths changed unexpectedly");
if (entries.length !== 42 || entries.filter(entry => entry.visibility === "public").length !== 37) throw new Error("Unexpected bounded registration totals");
for (const entry of entries) {
  if ((entry.baselineDisposition === "internalize") !== (entry.visibility === "internal" && entry.path.endsWith(":purgeAll"))) {
    if (!entry.path.includes("fetch") && !entry.path.includes("keywordSearchMemories")) throw new Error(`Disposition mismatch: ${entry.path}`);
  }
}

  sourceIntegrityPolicy: "Immutable full canonical source witnesses validated against normalized NFC/LF/trim content and actual semanticHash, then exact scoped rows/owners/full tuple/content/hash/lifecycle/timestamps/control tombstones reloaded after last crypto before current/history/hydration/CAS/bulk delivery; no hash after final reload",
