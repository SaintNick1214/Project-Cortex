import assert from 'node:assert/strict';
import schema from '../../../convex-dev/schema.ts';
import {defineTable} from 'convex/server';
import {v} from 'convex/values';
const duplicates=indexes=>indexes.flatMap((index,p)=>indexes.slice(p+1).filter(other=>JSON.stringify(index.fields)===JSON.stringify(other.fields)).map(other=>[index.indexDescriptor,other.indexDescriptor]));
const rejected=defineTable({tenantId:v.string(),memorySpaceId:v.string()}).index('first',['tenantId','memorySpaceId']).index('second',['tenantId','memorySpaceId']);
assert.deepEqual(duplicates(rejected[' indexes']()),[['first','second']]);
const tables=Object.entries(schema.tables);assert.equal(tables.length,31);
const rows=tables.map(([table,definition])=>({table,indexes:definition[' indexes'](),duplicates:duplicates(definition[' indexes']())}));
assert.ok(rows.every(row=>row.duplicates.length===0));
assert.deepEqual(rows.find(row=>row.table==='artifacts').indexes.filter(x=>x.fields.join(',')==='tenantId,memorySpaceId'),[{indexDescriptor:'by_tenant_space',fields:['tenantId','memorySpaceId']}]);
process.stdout.write(JSON.stringify({scope:'OFFLINE_INSTALLED_CONVEX_SCHEMA_REFLECTION_ONLY',tests:3,tableCount:31,normalIndexCount:rows.reduce((n,row)=>n+row.indexes.length,0),duplicates:0,adverseDuplicateCorrectlyDetected:true,rows})+'\n');
