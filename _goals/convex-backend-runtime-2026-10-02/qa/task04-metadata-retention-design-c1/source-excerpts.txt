237: // The AI Gateway reports the authoritative dollar cost of each request.
238: // @convex-dev/ai-sdk-provider surfaces it at
239: // `providerMetadata.convexGateway.cost` (USD); convert to nanodollars and pass
240: // it through as authoritative (finishRequest prefers it over the token-based
241: // estimate). No-op on older provider versions that don't surface it.
242: function extractGatewayCostNanos(result: any): number | undefined {
243:   const meta = result?.providerMetadata?.convexGateway;
244:   const costUsd = meta?.cost;
245:   if (typeof costUsd === "number" && Number.isFinite(costUsd) && costUsd >= 0) {
246:     return Math.round(costUsd * NANOS_PER_DOLLAR);
247:   }
248:   // Back-compat: honor an explicitly nano-denominated field if ever present.
249:   const costNanos = meta?.costNanos ?? result?.usage?.costNanos;
250:   if (typeof costNanos === "number" && Number.isFinite(costNanos) && costNanos >= 0) {
251:     return Math.round(costNanos);
252:   }
253:   return undefined;
254: }
