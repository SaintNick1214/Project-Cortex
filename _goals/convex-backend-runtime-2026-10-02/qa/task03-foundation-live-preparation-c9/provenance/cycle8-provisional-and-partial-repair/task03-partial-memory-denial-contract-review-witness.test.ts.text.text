import { describe, it, expect } from '@jest/globals';
import * as memories from '../../../convex-dev/memories';
import { harness, identity } from '../../../tests/unit/runtimeEndpointAuth/sharedfixtures';
import { dataDenied } from '../../../convex-dev/runtimeDataAuth';
import { authDenial } from '../../../_goals/convex-backend-runtime-2026-10-02/qa/task03-foundation-live-preparation-c8/scripts/native-engine.mjs';

const exact = { version: 1, code: 'FORBIDDEN', message: 'Access denied', retryable: false, outcome: 'not_dispatched' };
describe('unchanged partial-memory denial source witness', () => {
  for (const operation of ['updatePartialMemory', 'finalizePartialMemory'] as const) {
    for (const fixture of ['foreign-tenant', 'foreign-owner', 'missing'] as const) {
      it(`${operation}: ${fixture} returns legacy not-found despite no hydration/effect`, async () => {
        const h = harness(); const owner = await h.provisionScope();
        const id = await h.seedData('memories', owner, 'known-owner', { isPartial: true, tags: ['partial', 'streaming', 'keep'] });
        if (fixture !== 'missing') {
          await h.provisionScope({ tenantId: fixture === 'foreign-tenant' ? 'tenant-b' : 'tenant-a', subject: 'bob' });
          h.setIdentity({ ...identity, subject: 'bob' });
        }
        const before = structuredClone(h.db.rows); h.db.reads = []; h.db.attemptedWrites = [];
        let error: any;
        try { await h.invoke(memories[operation], { memoryId: fixture === 'missing' ? 'absent' : 'known-owner', content: 'attempt', metadata: {} }); }
        catch (caught) { error = caught; }
        expect(error.data).toBe('MEMORY_NOT_FOUND');
        expect(() => authDenial(undefined, error)).toThrow();
        expect(h.db.rows).toEqual(before); expect(h.db.attemptedWrites).toEqual([]);
        expect(h.db.reads.filter(read => read.table === 'memories')).toEqual([{ table: 'memories', index: 'by_runtime_scope_memoryId', ids: [] }]);
        expect(h.db.reads.flatMap(read => read.ids)).not.toContain(id);
      });
    }
    it(`${operation}: legitimate owner mutation retains content/revision/functionality`, async () => {
      const h = harness(); const owner = await h.provisionScope();
      const id = await h.seedData('memories', owner, 'known-owner', { isPartial: true, tags: ['partial', 'streaming', 'keep'] });
      expect(await h.invoke(memories[operation], { memoryId: 'known-owner', content: 'owner edit', metadata: { marker: 1 } })).toEqual({ success: true });
      const row = await h.db.get(id);
      expect(row).toMatchObject({ content: 'owner edit', version: 2, lineage: { sourceRevision: 2 }, partialMetadata: { marker: 1 } });
      if (operation === 'finalizePartialMemory') expect(row).toMatchObject({ isPartial: false, tags: ['keep'] });
      expect(h.db.rows.get('runtimeMemorySources')![0]).toMatchObject({ content: 'owner edit', lineage: { sourceRevision: 2 } });
    });
  }
  it('existing dataDenied helper emits exact admitted denial, without changing classifier', () => {
    let error: any; try { dataDenied(); } catch (caught) { error = caught; }
    expect(error.data).toEqual(exact); expect(() => authDenial(undefined, error)).not.toThrow();
  });
});
