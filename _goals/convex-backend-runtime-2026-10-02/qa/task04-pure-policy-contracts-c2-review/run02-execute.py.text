from pathlib import Path
import os,subprocess,json,datetime,hashlib,shutil
r=Path('/workspace/Project-Cortex');s=r/'work/resume/task04-pure-policy-c2-review/run02';q=r/'_goals/convex-backend-runtime-2026-10-02/qa/task04-pure-policy-contracts-c2-review'
