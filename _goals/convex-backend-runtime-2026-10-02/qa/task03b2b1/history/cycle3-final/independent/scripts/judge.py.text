def verify(name,base,records):
 rows=[]
 if isinstance(records,list):items=[(r.get('path',r.get('relativePath')),r) for r in records]
 else:items=records.items()
 for path,want in items:
  p=pathlib.Path(path);p=p if p.is_absolute() else base/p
  got=record(p) if p.is_file() else None
  if isinstance(want,str):same=bool(got and got['sha256']==want)
  else:same=bool(got and got['sha256']==want.get('sha256',want.get('hash')) and (want.get('bytes',want.get('size')) is None or got['bytes']==want.get('bytes',want.get('size'))))
  rows.append({'path':str(p),'wanted':want,'actual':got,'matches':same})
 d={'records':len(rows),'mismatchCount':sum(not r['matches'] for r in rows),'rows':rows};(OUT/'raw'/f'{name}.json').write_text(json.dumps(d,indent=2)+'\n');print(name,d['records'],d['mismatchCount']);return d
