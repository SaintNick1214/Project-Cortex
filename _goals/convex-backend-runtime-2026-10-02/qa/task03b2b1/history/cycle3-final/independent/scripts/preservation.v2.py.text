  else:
   matches=[p for p in ready.rglob('*') if p.is_file() and record(p)=={'sha256':r['sha256'],'bytes':r['bytes']}]

     mirror=Q/'history/cycle2-final/candidate/source'/pathlib.Path(str(src)+'.text')
     if mirror.is_file() and record(mirror)=={'sha256':r['sha256'],'bytes':r['bytes']}:matches=[mirror]
    assert matches,src
    dest=matches[0]
  resolved[str(dest)]=r
