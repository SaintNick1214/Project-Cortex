import pathlib, json, hashlib, subprocess, tarfile, datetime, os
root=pathlib.Path('/workspace/Project-Cortex')
out=pathlib.Path('/tmp/task03c2-live-harness-review1-7m04c8w4')
qa=root/'_goals/convex-backend-runtime-2026-10-02/qa/task03c2-live-fixture'
scratch=root/'work/backend-runtime/task03c2-live-fixture'
sha=lambda b:hashlib.sha256(b).hexdigest()
read=lambda p:json.loads(p.read_text())
accepted='4fb3482c5e06fcab12ab8513b242a066ef761f2f'
backend='c92d548368791386c62f39895e381ef6bde2dc86'
def gitbytes(commit,path): return subprocess.check_output(['git','show',f'{commit}:{path}'],cwd=root)
checks=[]
def check(name,ok,detail=None):
    checks.append({'name':name,'pass':bool(ok),'detail':detail})
freeze=read(qa/'source-freeze.json')
check('freeze-timestamp',freeze['frozenAt']=='2026-10-03T11:58:44.330089+00:00')
check('all-frozen-bytes',all(sha((qa/e['path']).read_bytes())==e['sha256'] for e in freeze['files']),len(freeze['files']))
prov=read(qa/'source-provenance.json')
fixture=[]
for e in prov['copied']:
    actual=(qa/e['fixture']).read_bytes()
    fixture.append({'path':e['fixture'],'sha256':sha(actual),'accepted':actual==gitbytes(accepted,e['source']),'backend':actual==gitbytes(backend,e['source'])})
check('14-fixture-accepted-bytes',len(fixture)==14 and all(e['accepted'] and e['backend'] for e in fixture),fixture)
sdk=read(qa/'sdk-provenance.json')
src=[]
for e in sdk['files']:
    p=scratch/e['path']; b=p.read_bytes()
    if e['path'].startswith('sdk/') and not e['path'].startswith('sdk/dist/'):
        orig=e['path'][4:]
        src.append({'path':orig,'sha256':sha(b),'accepted':b==gitbytes(accepted,orig)})
    check('sdk-file:'+e['path'],sha(b)==e['sha256'])
check('129-sdk-exact',len(src)==129 and all(e['accepted'] for e in src),src)
archive=read(qa/'archive-preservation.json')
check('107-archives-unchanged',len(archive)==107 and all(sha((root/e['path']).read_bytes())==e['sha256'] for e in archive),len(archive))
for manifest,base in [('file-inventory.json',root),('private-file-inventory.json',root)]:
    data=read(qa/manifest)
    failures=[]
    for e in data:
        p=base/e['path']
        if not p.exists() or sha(p.read_bytes())!=e['sha256']: failures.append(e['path'])
    check(manifest+'-hashes',not failures,{'entries':len(data),'failures':failures})
pack=scratch/sdk['package']['path']
check('pack-hash',sha(pack.read_bytes())==sdk['package']['sha256'],sha(pack.read_bytes()))
with tarfile.open(pack) as t:
    entries=t.getmembers()
    check('pack-53-entries',len(entries)==53,len(entries))
    diffs=[e.name for e in entries if not e.isfile() or t.extractfile(e).read()!=(scratch/'packed-sdk'/e.name).read_bytes()]
    check('pack-extraction-bytes',not diffs,diffs)
    pkg=json.loads(t.extractfile('package/package.json').read())
    check('pack-entry-export',set(pkg['exports'])=={'.'},list(pkg['exports']))
matrix=read(qa/'service-outcomes.json')
check('41-unexecuted',matrix['discovered']==41 and matrix['executed']==0 and len(matrix['cases'])==41 and all(e['status']=='UNEXECUTED' for e in matrix['cases']))
history=root/'_goals/convex-backend-runtime-2026-10-02/qa/ci-followups/history/metadata-cycle3-pre-lint'
hist=read(history/'provenance.json')
histresults=[]
for e in [hist['helper'],*hist['historicalManifests']]:
    b=(root/e['archivePath']).read_bytes(); orig=gitbytes(backend,e['originalPath'])
    histresults.append({'path':e['archivePath'],'sha256':sha(b),'gitExact':b==orig,'currentOriginalUnchanged': e is hist['helper'] or (root/e['originalPath']).read_bytes()==orig})
check('ancillary-history-exact',all(e['gitExact'] and e['currentOriginalUnchanged'] for e in histresults),histresults)
helper=(root/hist['helper']['originalPath']).read_text()
old=gitbytes(backend,hist['helper']['originalPath']).decode()
expected=old.replace('const omitLine = ({ line: _line, ...row }) => row;','const omitLine = (entry) => {\n  const row = { ...entry };\n  delete row.line;\n  return row;\n};')
check('helper-only-equivalent-edit',helper==expected,sha(helper.encode()))
statpaths=[scratch,root/'work/backend-runtime/qualification/jwt-private.pem']
metadata=[]
for p in statpaths:
    s=p.lstat();metadata.append({'path':str(p.relative_to(root)),'mode':oct(s.st_mode&0o777),'uid':s.st_uid,'nlink':s.st_nlink,'symlink':p.is_symlink()})
check('private-root-and-key-metadata',metadata[0]['mode']=='0o700' and metadata[1]['mode']=='0o600' and metadata[1]['uid']==os.getuid() and metadata[1]['nlink']==1 and all(not e['symlink'] for e in metadata),metadata)
check('no-created-live-target',not (scratch/'target.json').exists() and not (scratch/'deploy.env').exists())
hygiene=[]
import re
for p in qa.rglob('*'):
    if p.is_file():
        b=p.read_bytes()
        scrub=b.replace(b'dev:offline-fixture-123|OFFLINE_STANDIN_NOT_A_CREDENTIAL',b'PUBLIC_OFFLINE_STANDIN')
        if re.search(rb'-----BEGIN (?:RSA |EC |OPENSSH )?PRIVATE KEY-----|\beyJ[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+|\b(?:dev|prod):[a-z0-9-]+\|[A-Za-z0-9_=-]{20,}',scrub): hygiene.append(str(p.relative_to(qa)))
check('public-artifacts-no-secret-patterns',not hygiene,hygiene)
result={'observedAt':datetime.datetime.now(datetime.timezone.utc).isoformat(),'checks':checks,'allPass':all(e['pass'] for e in checks)}
(out/'integrity.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps({'allPass':result['allPass'],'checks':len(checks),'failures':[e for e in checks if not e['pass']]}))
