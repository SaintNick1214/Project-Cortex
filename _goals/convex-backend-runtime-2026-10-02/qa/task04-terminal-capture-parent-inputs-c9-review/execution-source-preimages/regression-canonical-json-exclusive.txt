function canonical(path,base,privateFile=false){if(path!==resolve(path)||!path.startsWith(base+sep)&&path!==base)throw Error('QA_PATH');let part=sep;
 for(const component of path.split(sep).filter(Boolean)){part=resolve(part,component);const s=lstatSync(part);if(s.isSymbolicLink()||realpathSync(part)!==part)throw Error('QA_PATH');}
 const s=lstatSync(path);if(s.isFile()&&s.nlink!==1)throw Error('QA_LINK');if(privateFile&&(!s.isFile()||s.uid!==process.getuid()||(s.mode&511)!==384))throw Error('QA_PRIVATE');return path;}
function json(path,base=root,privateFile=false){return JSON.parse(readFileSync(canonical(path,base,privateFile),'utf8'));}
function exclusive(path,value,privateFile=true,maxBytes=limits.resultBytes,control){if(control!==undefined&&(!control||Object.keys(control).some(key=>!['deadlineMs','beforeCreate','observe'].includes(key))||!Number.isSafeInteger(control.deadlineMs)||control.deadlineMs<1||typeof control.beforeCreate!=='function'||typeof control.observe!=='function'))throw Error('QA_EXCLUSIVE_CONTROL');canonical(dirname(path),repo);if(control&&privateFile){const parent=lstatSync(dirname(path));if(!parent.isDirectory()||parent.uid!==process.getuid()||(parent.mode&511)!==448)throw Error('QA_PRIVATE_PARENT');}const encoded=Buffer.from(JSON.stringify(value,null,2)+'\n');if(encoded.length>maxBytes)throw Error('QA_BYTES');if(control)control.beforeCreate();if(control&&Date.now()>=control.deadlineMs)throw Error('QA_EXCLUSIVE_DEADLINE');const fd=openSync(path,'wx',privateFile?0o600:0o644);let primary,hasPrimary=false;const observe=stage=>{if(control)control.observe(stage);};try{observe('created');let offset=0;while(offset<encoded.length){const wrote=writeSync(fd,encoded,offset,encoded.length-offset);if(!Number.isSafeInteger(wrote)||wrote<1||wrote>encoded.length-offset)throw Error('QA_EXCLUSIVE_SHORT_WRITE');offset+=wrote;}observe('written');fsyncSync(fd);observe('durable');}catch(error){primary=error;hasPrimary=true;}let closed=false;for(let attempt=0;attempt<2&&!closed;attempt++){try{closeSync(fd);closed=true;}catch(error){if(!hasPrimary){primary=error;hasPrimary=true;}if(attempt===1&&error?.code==='EBADF')closed=true;}}if(closed)try{observe('closed');}catch(error){if(!hasPrimary){primary=error;hasPrimary=true;}}if(hasPrimary)throw primary;}
