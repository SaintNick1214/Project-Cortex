/** NEW additive nonpaid native outcomes. Actual native session required; no fake accounting sink. */
import assert from 'node:assert/strict';
import {canonical,request} from './original-b-cases.mjs';
import {qualifierDeadline} from './guard.mjs';
import {gatewayNative} from './gateway-native-composition.mjs';
export {canonical,request};
export const definitions=['known-drift-40','known-drift-150','unknown-drift-terminal','unknown-lifetime','token-input-unknown','token-input-known40','token-output-unknown','token-output-known40','refinement-add-absent-model','refinement-add-absent-tokens','captured-parity-settle','captured-parity-recover','captured-parity-operator',...['initial','final'].flatMap(source=>['model','token'].flatMap(axis=>['pending','attached'].map(privacy=>'defensive-'+source+'-'+axis+'-'+privacy))),'gateway-native-drift-40','gateway-native-drift-150','gateway-native-healthy-attachment','late-drift-refinement-revoked-cancelled','late-drift-refinement-original-window'].map(id=>[id,id]);
export async function scenario(s,id){
 const call=(name,args)=>s.rpc(name==='resolve'?'query':'mutation','runtimeModelPolicy:'+name,args);
 const snapshot=()=>s.rpc('query','qualificationOwned:snapshot',{run:s.runId});
 const same=(a,b)=>assert.equal(canonical(a),canonical(b),'QA_WHOLE_STATE_DIFFERENT');
 const denied=async fn=>{const before=await snapshot();await assert.rejects(fn,e=>Boolean(e?.data?.code));same(before,await snapshot());};
 const duplicate=async(fn,expected)=>{const before=await snapshot();const result=await fn();if(expected)same(result,expected);same(before,await snapshot());return result;};
 const seed=await s.rpc('mutation','qualificationOwned:seed',{run:s.runId,caseId:id,subject:s.runId+'-'+id});
 await call('installDeploymentPolicy',{manifestId:'qa-task04-synthetic-v1',expectedRevision:1});
 const b=await call('ensureBudget',{reference:seed.reference,operationKey:'agent.respond',requestId:id,requestCanonical:canonical({text:'hello'})});
 const args={reference:seed.reference,operationKey:'agent.respond',runBudgetId:b.runBudgetId,semanticId:'semantic',ordinal:0,requestCanonical:canonical(request())};
 // Gateway-source arms own the actual registered admission/checkpoint; never pre-admit through a fake adapter.
 if(id.startsWith('gateway-native-')){await gatewayNative(s,id,{seed,b,args,call,snapshot,duplicate,denied});return;}
 const a=await call('admit',args);assert.equal(a.status,'reserved');await call('checkpoint',{reference:seed.reference,attemptId:a.attemptId,requestCanonical:a.requestCanonical});
 const initialRows=await snapshot(),attempt=initialRows.find(x=>x.id===a.attemptId).row,selected=JSON.parse(attempt.snapshotCanonical);
 const bound=rows=>{const t=rows.find(x=>x.id===a.attemptId).row,o=rows.find(x=>x.id===t.operationDocumentId).row,accounts=[t.tenantAccountId,t.runAccountId].map(key=>rows.find(x=>x.id===key).row);return{t,o,accounts};};
 const monetary=state=>({operation:{reservedUnits:state.o.reservedUnits,uncertainUnits:state.o.uncertainUnits,settledUnits:state.o.settledUnits},accounts:state.accounts.map(r=>({id:r._id,accountKey:r.accountKey,windowStart:r.windowStart,windowEnd:r.windowEnd,reservedUnits:r.reservedUnits,uncertainUnits:r.uncertainUnits,settledUnits:r.settledUnits,activeConcurrency:r.activeConcurrency})),slotReleased:state.t.slotReleased});
 const observe=()=>({actualModel:selected.modelId,usage:{inputTokens:attempt.measuredInputTokens,outputTokens:attempt.maximumOutputTokens}});
 const receipt=(cost,extra={})=>canonical({version:1,dispatchIdentity:attempt.dispatchIdentity,receiptId:'qa-terminal-'+id,modelId:selected.modelId,nativeInterface:selected.nativeInterface,outcome:'confirmed',proof:'provider-terminal-v1',costSource:cost===undefined?'unavailable':'gateway-aggregate-usd-v1',...(cost===undefined?{}:{aggregateCostUsd:'0.000'+String(cost).padStart(3,'0')}),...extra});
 const capture=encoded=>call('captureTerminalEvidence',{attemptId:a.attemptId,dispatchIdentity:attempt.dispatchIdentity,receiptCanonical:encoded});
 const attach=evidenceId=>call('attachTerminalResult',{reference:seed.reference,attemptId:a.attemptId,evidenceId,privateResultCanonical:canonical({text:'answer'})});
 const control=variant=>s.rpc('mutation','qualificationCapture:partialMarker',{run:s.runId,attemptId:a.attemptId,variant});
 const origin=name=>s.rpc('mutation','qualificationCapture:'+name,{run:s.runId,reference:seed.reference,attemptId:a.attemptId});
 const fault=name=>s.rpc('mutation','qualificationOwned:change',{run:s.runId,reference:seed.reference,attemptId:a.attemptId,operationId:a.operationId,fault:name});
 const accounting=(rows,cost,unknown=0,slots=0,violation=true)=>{const state=bound(rows);for(const r of [state.o,...state.accounts]){assert.equal(r.reservedUnits,0);assert.equal(r.uncertainUnits,unknown);assert.equal(r.settledUnits,cost);assert.equal(r.overrun,violation);}for(const r of state.accounts)assert.equal(r.activeConcurrency,slots);assert.equal(state.t.slotReleased,slots===0);return state;};
 const closed=state=>{assert.equal(state.t.resultAttachmentState,'closed');assert.equal(state.t.privateResultCanonical,undefined);assert.equal(state.t.outputVisible,false);};
 const guarded=async()=>{await denied(()=>call('ensureBudget',{reference:seed.reference,operationKey:'agent.respond',requestId:'blocked-successor',requestCanonical:canonical({text:'hello'})}));await denied(()=>call('admit',{...args,semanticId:'child',parentOperationId:a.operationId,ordinal:1}));await denied(()=>call('admitFallback',{reference:seed.reference,attemptId:a.attemptId,fallbackModelId:selected.modelId}));const result=await duplicate(()=>call('checkpoint',{reference:seed.reference,attemptId:a.attemptId,requestCanonical:a.requestCanonical}));assert.equal(result.dispatchPermit,false);assert.equal(result.replayAllowed,false);};
 if(id==='unknown-lifetime'){
  const unsupported=canonical({...JSON.parse(receipt(undefined,{actualModel:'qualification/drift-model',usage:{inputTokens:attempt.measuredInputTokens+1}})),proof:'unqualified'});await denied(()=>capture(unsupported));const pre=bound(await snapshot());assert.equal(pre.t.terminalEvidenceHash,undefined);for(const r of pre.accounts){assert.equal(r.reservedUnits,100);assert.equal(r.activeConcurrency,1);}await call('recover',{reference:seed.adminReference,attemptId:a.attemptId});const state=accounting(await snapshot(),0,100,1,false);assert.equal(state.t.terminalEvidenceHash,undefined);const replay=await call('checkpoint',{reference:seed.reference,attemptId:a.attemptId,requestCanonical:a.requestCanonical});assert.equal(replay.dispatchPermit,false);assert.equal(replay.replayAllowed,false);return;
 }
 if(id.startsWith('defensive-')){
  const healthy=receipt(40,observe()),first=await capture(healthy);const baseline=bound(await snapshot());for(const r of [baseline.o,...baseline.accounts])assert.equal(r.overrun,false);assert.equal(baseline.t.resultAttachmentState,'pending');
  const variant=id.slice('defensive-'.length),changed=await control(variant);let currentEvidence=changed.evidenceId;let before=bound(await snapshot());for(const r of [before.o,...before.accounts])assert.equal(r.overrun,false);assert.equal(before.t.outputVisible,true);assert.equal(before.t.privateResultCanonical,canonical({text:'answer'}));assert.equal(before.t.resultAttachmentState,variant.endsWith('attached')?'attached':'pending');
  same(await attach(changed.evidenceId),{attached:false,resultWithheld:true});let after=bound(await snapshot());same(monetary(before),monetary(after));await duplicate(()=>attach(changed.evidenceId),{attached:false,resultWithheld:true});
  if(variant.includes('-model-')){
   const isolated=await control('attempt-only-'+(variant.endsWith('attached')?'attached':'pending'));currentEvidence=isolated.evidenceId;before=bound(await snapshot());const i=JSON.parse(before.t.terminalEvidenceCanonical),f=JSON.parse(before.t.receiptCanonical);assert.equal(i.actualModel,selected.modelId);assert.equal(f.actualModel,selected.modelId);assert.equal(before.t.actualModel,'qualification/drift-model');for(const r of [before.o,...before.accounts])assert.equal(r.overrun,false);same(await attach(isolated.evidenceId),{attached:false,resultWithheld:true});after=bound(await snapshot());same(monetary(before),monetary(after));await duplicate(()=>attach(isolated.evidenceId),{attached:false,resultWithheld:true});
  }
  await origin('renewOrigin');const duplicateBefore=await snapshot();const returned=await call('recover',{reference:seed.adminReference,attemptId:a.attemptId,receiptCanonical:receipt(40,observe())});assert.equal(returned.resultWithheld,true);same(duplicateBefore,await snapshot());same(await attach(currentEvidence),{attached:false,resultWithheld:true});return;
 }
 let facts={actualModel:'qualification/drift-model'},firstCost=id==='known-drift-40'?40:id==='known-drift-150'?150:undefined;
 if(id.startsWith('token-')){const input=id.startsWith('token-input-');facts=observe();facts.usage[input?'inputTokens':'outputTokens']=(input?attempt.measuredInputTokens:attempt.maximumOutputTokens)+1;firstCost=id.endsWith('-known40')?40:undefined;}
 if(id.startsWith('refinement-add-absent-'))facts={};
 const initial=receipt(firstCost,facts),first=await capture(initial),firstRows=await snapshot(),state=accounting(firstRows,firstCost??0,firstCost===undefined?100:0,0,!id.startsWith('refinement-add-absent-'));closed(state);assert.equal(state.t.terminalEvidenceCanonical,initial);assert.equal(state.t.actualModel,facts.actualModel);
 if(firstCost!==undefined){await duplicate(()=>capture(initial),first);same(await attach(first.evidenceId),{attached:false,resultWithheld:true});await guarded();return;}
 let finalFacts=facts;
 if(id==='refinement-add-absent-model')finalFacts={actualModel:'qualification/drift-model'};
 if(id==='refinement-add-absent-tokens')finalFacts={actualModel:selected.modelId,usage:{inputTokens:attempt.measuredInputTokens+1,outputTokens:attempt.maximumOutputTokens+1}};
 let cost=id.startsWith('captured-parity-')||id.startsWith('late-drift-')?150:40;
 const final=receipt(cost,finalFacts);
 if(id==='late-drift-refinement-revoked-cancelled'){await origin('revokeOrigin');const live=await snapshot();assert.equal(live.filter(x=>x.table==='runtimeAuthTombstones').length,0);assert.equal(live.filter(x=>x.table==='runtimeAuthScopes'&&x.row.tenantId===seed.reference.tenantId).length,2);await fault('parent-cancel');}
 if(id==='late-drift-refinement-original-window'){const end=Math.max(...state.accounts.map(r=>r.windowEnd));let waits=0;while(Date.now()<=end+1000){if(++waits>248||Date.now()>=qualifierDeadline())throw Error('QA_TOTAL_TIME_BUDGET');await new Promise(done=>setTimeout(done,250));}await denied(()=>call('ensureBudget',{reference:seed.reference,operationKey:'agent.respond',requestId:'current-before',requestCanonical:canonical({text:'hello'})}));}
 let result;
 if(id.startsWith('captured-parity-')){
  const route=id==='captured-parity-settle'?'settle':id==='captured-parity-recover'?'recover':'reconcileDeploymentLiability';const routeArgs={attemptId:a.attemptId,receiptCanonical:receipt(cost,{...finalFacts,privateResultCanonical:canonical({text:'secret'})}),...(route==='reconcileDeploymentLiability'?{}:{reference:route==='recover'?seed.adminReference:seed.reference})};
  if(route==='reconcileDeploymentLiability'){await denied(()=>call(route,routeArgs));await fault('tenant-delete');}else await denied(()=>call(route,{...routeArgs,reference:{...routeArgs.reference,grantVersion:routeArgs.reference.grantVersion+1}}));
  result=await call(route,routeArgs);assert.equal(result.resultWithheld,true);await duplicate(()=>call(route,routeArgs));await denied(()=>call(route,{...routeArgs,receiptCanonical:receipt(cost,{actualModel:'qualification/changed-model'})}));
 }else result=await capture(final);
 const finalRows=await snapshot(),post=accounting(finalRows,cost);closed(post);for(const key of ['terminalEvidenceCanonical','terminalEvidenceHash'])assert.equal(post.t[key],state.t[key]);for(let index=0;index<2;index++)for(const key of ['_id','accountKey','windowStart','windowEnd'])assert.equal(post.accounts[index][key],state.accounts[index][key]);assert.equal(post.t.actualModel,finalFacts.actualModel);assert.equal(post.t.receiptCanonical,final);
 const newest=await duplicate(()=>capture(final));assert.equal(newest.evidenceId,first.evidenceId);await duplicate(()=>capture(initial),newest);same(await attach(first.evidenceId),{attached:false,resultWithheld:true});
 if(id.startsWith('refinement-add-absent-'))await denied(()=>capture(receipt(cost,{actualModel:selected.modelId})));
 if(id==='late-drift-refinement-original-window'){await denied(()=>call('ensureBudget',{reference:seed.reference,operationKey:'agent.respond',requestId:'current-after',requestCanonical:canonical({text:'hello'})}));assert.equal(finalRows.filter(x=>x.table==='runtimeBudgetAccounts'&&x.row.tenantId===seed.reference.tenantId&&!state.accounts.some(r=>r._id===x.id)).length,0);}
}
